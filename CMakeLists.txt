cmake_minimum_required(VERSION 3.21)
project(cra_browser_linux LANGUAGES C CXX)

set(CEF_ROOT "" CACHE PATH "Extracted Linux CEF minimal distribution")
if(NOT EXISTS "${CEF_ROOT}/cmake/FindCEF.cmake")
  message(FATAL_ERROR "Set CEF_ROOT to an extracted Linux CEF distribution")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
list(REMOVE_ITEM CEF_LINKER_FLAGS "-Wl,-rpath,.")
# Prefer the distribution hardening level when its build flags set one.
list(REMOVE_ITEM CEF_COMPILER_FLAGS_RELEASE "-U_FORTIFY_SOURCE" "-D_FORTIFY_SOURCE=2")
foreach(language IN ITEMS C CXX)
  set(flags "${CMAKE_${language}_FLAGS} ${CMAKE_${language}_FLAGS_RELEASE}")
  if(NOT flags MATCHES "(^|[ ,])-D_FORTIFY_SOURCE(=|[ ,]|$)")
    add_compile_options("$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:${language}>>:-D_FORTIFY_SOURCE=2>")
  endif()
endforeach()
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")

add_executable(cra-browser src/main.cc src/browser.cc src/session.cc)
SET_EXECUTABLE_TARGET_PROPERTIES(cra-browser)
target_compile_features(cra-browser PRIVATE cxx_std_20)
target_link_libraries(cra-browser PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
set_target_properties(cra-browser PROPERTIES BUILD_RPATH "$ORIGIN;${CEF_BINARY_DIR}" INSTALL_RPATH "$ORIGIN")

install(TARGETS cra-browser RUNTIME DESTINATION lib/cra-browser-cef)
install(DIRECTORY "${CEF_BINARY_DIR}/" "${CEF_RESOURCE_DIR}/" DESTINATION lib/cra-browser-cef
        USE_SOURCE_PERMISSIONS PATTERN "*.a" EXCLUDE )
install(PROGRAMS scripts/cra-browser DESTINATION bin RENAME cra-browser-cef)
install(FILES data/app.crabrowser.CraCEF.desktop DESTINATION share/applications)
install(FILES data/app.crabrowser.CraCEF.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION share/licenses/cra-browser-cef)
install(FILES "${CEF_ROOT}/LICENSE.txt" DESTINATION share/licenses/cra-browser-cef RENAME CEF-LICENSE.txt)

include(CTest)
if(BUILD_TESTING)
  add_executable(session-test tests/session_test.cc src/session.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(session-test)
  target_include_directories(session-test PRIVATE src)
  target_compile_features(session-test PRIVATE cxx_std_20)
  target_link_libraries(session-test PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(session-test PROPERTIES BUILD_RPATH "${CEF_BINARY_DIR}")
  add_test(NAME session-round-trip COMMAND session-test)
endif()
